defineFunction(CalculateVehicleDemand, float)
{
    Vehicle_GetNumberOfVehicles(m);
    vehicles_in_depot = 0;
    vehicles_num_trucks = 0;
    vehicles_num_trains = 0;
    vehicles_num_ships = 0;
    vehicles_num_airplanes = 0;
    demand_vehicles = 0.0;

    for (i=0, i<m, i=i+1)
    {
        vehicle.GetDataByIndex(i);
        vehicle_cost = 0.0;
        if(vehicle.nTransportType ? RESOURCE_TRANSPORT_PASSANGER)
        {
            vehicle_cost = vehicle.fResourceCapacity / 10.0;
        }
        elseif(vehicle.nTransportType > -1 & vehicle.nTransportType < RESOURCE_TRANSPORT_NUM)
        {
            vehicle_cost = vehicle.fResourceCapacity;
        }

        if (vehicle.nVehicleType > 0 & vehicle.nVehicleType < 3)
        {
            vehicles_num_trucks = vehicles_num_trucks + 1;
            if(vehicle.nSkill[0] ? VEHICLESKILL_SEEDING) { vehicle_cost = BASE_COST_SEEDING; }
            elseif(vehicle.nSkill[0] ? VEHICLESKILL_HARVESTING) { vehicle_cost = BASE_COST_HARVESTING; }
            elseif(vehicle.nSkill[0] ? VEHICLESKILL_SNOWPLOW) { vehicle_cost = BASE_COST_SNOWPLOWING; }
            elseif(vehicle.nSkill[0] ? VEHICLESKILL_PERSONAL) { vehicle_cost = BASE_COST_PERSONAL; }
            else() { vehicle_cost = BASE_COST_ROAD + vehicle_cost * COST_ROAD_FACTOR; }
        }
        elseif(vehicle.nVehicleType < 6)
        {
            vehicles_num_trains = vehicles_num_trains + 1;
            vehicle_cost = BASE_COST_RAIL + COST_RAIL_FACTOR * vehicle_cost;
        }
        elseif(vehicle.nVehicleType ? VEHICLETYPE_SHIP)
        {
            vehicles_num_ships = vehicles_num_ships + 1;
            vehicle_cost = vehicle_cost + BASE_COST_SHIP + COST_SHIP_FACTOR * vehicle_cost;
        }
        elseif(vehicle.nVehicleType ? VEHICLETYPE_AIRPLANE | vehicle.nVehicleType ? VEHICLETYPE_HELICOPTER)
        {
            vehicles_num_airplanes = vehicles_num_airplanes + 1; 
            vehicle_cost = vehicle_cost + BASE_COST_AIR + COST_AIR_FACTOR * vehicle_cost;
        }

        if(vehicle.nBuilding_CurrentID ? vehicle.nBuilding_HomeWorkplaceID & vehicle.nBuilding_CurrentID > 0)
        {
            debug_workplace_id = vehicle.nBuilding_HomeWorkplaceID;
            vehicle_cost = vehicle_cost * VEHICLE_IDLE_COST;
            vehicles_in_depot = vehicles_in_depot + 1;
        }

        demand_vehicles = demand_vehicles + vehicle_cost;
    }
    return(demand_vehicles);
}

defineFunction(CalculateCivicDemand, float)
{
    population = 0;
    for (i=0, i<n, i=i+1)
    {
        building.GetDataByIndex(i);
        if(building.nType ? BUILDINGTYPE_LIVING & building.fPercFinished > 0.9999)
        {
            Building_IsOldTownActive(i, is_active);
            population = population + is_active * is_active * building.nWorkersNum;
        }
    }
    
    demand = 0.0;
    if(!(population > GRACE_CITIZENS))
    {
        demand = population * INTERNAL_DEMAND_FACTOR * GRACE_FACTOR;
    }
    else()
    {
        demand = INTERNAL_DEMAND_FACTOR * (GRACE_CITIZENS * GRACE_FACTOR + population - GRACE_CITIZENS);
    }
    return(demand);
}

defineFunction(CalculateSupply, float)
{
    supply = 0.0;
    for (i=0, i<n, i=i+1)
    {
        bld_services = 0.0;
        building.GetDataByIndex(i);

        if (!(building.nSubType ? BUILDINGSUBTYPE_ROAD) & building.fPercFinished > 0.9999)
        {
            if(building.nType ? BUILDINGTYPE_CITYHALL)
            {
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    service_scale = PRIMARY_ED_FACTOR;
                    if (worker.fEducation>1.999) { service_scale = UNIVERSITY_ED_FACTOR; }
                    bld_services = bld_services + service_scale;
                }
            }
            if(building.nType ? BUILDINGTYPE_UNIVERSITY & building.nSubType ? BUILDINGSUBTYPE_SOVIET)
            {
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    if (worker.bWorkplaceAsProfesor ? 1) { bld_services = bld_services + HQ_FACTOR; }
                }
            }
            if(building.nType ? BUILDINGTYPE_SECRET_POLICE & building.nVehiclesWorkingNum < 1)
            {
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    service_scale = PRIMARY_ED_FACTOR;
                    if (worker.fEducation>1.999) { service_scale = UNIVERSITY_ED_FACTOR; }
                    bld_services = bld_services + service_scale;
                }
            }
            if (VEHICLE_DIFFICULTY > 0.0 & building.nType ? BUILDINGTYPE_AIRPLANE_TOWER)
            {
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    service_scale = PRIMARY_ED_FACTOR;
                    if (worker.fEducation>1.999) { service_scale = UNIVERSITY_ED_FACTOR; }
                    bld_services = bld_services + service_scale;
                }
            }
        }
        if(building.fEletric_Voltage < 0.9 & game.EletricityLevel > 0) { bld_services = bld_services * NO_POWER_FACTOR; }
        if(building.fWater_CurrentDrinkAmount < 0.001 & game.WaterManagement > 0) { bld_services = bld_services * NO_WATER_FACTOR; }
        
        supply = supply + bld_services;
    }
    return(supply);
}

                supply_party_hq = supply_party_hq + bld_services;

            {
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    service_scale = PERMIT_SUPPLY_SECRET_POLICE;
                    if (worker.fEducation>1.999) { service_scale = PERMIT_SUPPLY_SECRET_POLICE * 1.5; }
                    bld_services = bld_services + service_scale;
                }
                if(building.fEletric_Voltage < 0.9 & game.EletricityLevel > 0) { bld_services = bld_services * NO_POWER_FACTOR; }
                if(building.fWater_CurrentDrinkAmount < 0.001 & game.WaterManagement > 0) { bld_services = bld_services * NO_WATER_FACTOR; }
                supply_civic = supply_civic + bld_services;
                supply_secret_police = supply_secret_police + bld_services;
            }

                for (j=0, j<building.nWorkersNum, j=j+1)

defineFunction(ProcessPermitPenalties, void)
{
    if (daily_permit_demand > 0.0)
    {
        permit_deficit_ratio = (daily_unmitigated_permit_deficit / daily_permit_demand) * 100.0; 
    }
    else()
    {
        permit_deficit_ratio = 0.0;
    }

    Building_GetNumberOfBuildings(n);
    
    for (i = 0, i < n, i = i + 1)
    {
        building.GetDataByIndex(i);
        Building_GetCharTag(i, tag_buf);

        strcmp(tag_buf, tag_empty, cmp_res);
        if (cmp_res ? 0)
        {
            if (building.fPercFinished > 0.9999) { Building_SetTagChar(tag_clean, i); }
            else() { Building_SetTagChar(tag_under_construction, i); }
        }

        if (building.bConstructionActive ? 1 & daily_unmitigated_permit_deficit > 0.0)
        {
            Random(roll);
            roll = roll % 10000;
            float_roll = roll * 1.0; 
            
            if (float_roll < (permit_deficit_ratio * PERMIT_FIRE_CHANCE_MULT) & building.bFireEnabled ? 1)
            {
                Building_StartFire(i);
            }
        }

        if (building.fPercFinished > 0.9999)
        {
            strcmp(tag_buf, tag_under_construction, cmp_res);
            if (cmp_res ? 0)
            {
                if (daily_unmitigated_permit_deficit > 0.0) { Building_SetTagChar(tag_rushed, i); }
                else() { Building_SetTagChar(tag_clean, i); }
            }

            strcmp(tag_buf, tag_rushed, cmp_res);
            if (cmp_res ? 0)
            {
                for (j = 0, j < building.nWorkersNum, j = j + 1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);

                    if (worker.fStatusHealth > 0.1) { Person_SetStatus(worker_id, 2, worker.fStatusHealth - PERMIT_PENALTY_HEALTH_DRAIN); }
                    if (worker.fStatusHappiness > 0.1) { Person_SetStatus(worker_id, 0, worker.fStatusHappiness - PERMIT_PENALTY_HAPP_DRAIN); }
                }
            }
        }
    }
    returnVoid();
}

defineFunction(Price, float, int:currency, float:excess) 
{
    if (BUREAUCRACY_ENABLED > 0) { price_inertia = admin_pool + BASE_FOREIGN_SUPPLY_DEMAND + FOREIGN_SUPPLY_DEMAND_FACTOR * population; }
    else() { price_inertia = supply + BASE_FOREIGN_SUPPLY_DEMAND + FOREIGN_SUPPLY_DEMAND_FACTOR * population; }
    
    factor = 1.0 - excess / price_inertia;
    if(excess > 0.0) { price = factor * (currency * PRICE_SELL_RUB + (1 - currency) * PRICE_SELL_USD) / (100 ^ NUMBER_OF_PRICE_DECREASES); }
    else() { price = factor * (currency * PRICE_BUY_RUB + (1 - currency) * PRICE_BUY_USD) / (100 ^ NUMBER_OF_PRICE_DECREASES); }
    return(price);
}

defineFunction(MakePeopleAngry, void, float:unpaid_services)
{
    for(i=0, i<population, i=i+1)
    {
        worker.GetDataByIndex(i);
        Person_SetStatus(i, 0, MaxFloat(worker.fStatusHappiness - unpaid_services * UNHAPPINESS_FACTOR / 20.0, 0.0));
        Person_SetStatus(i, 3, MaxFloat(worker.fStatusSoviet - unpaid_services * LOYALTY_FACTOR / 20.0, 0.0));
        if(game.CrimeEnabled > 0) { Person_SetStatus(i, 10, MinFloat(worker.fStatusCrime + unpaid_services * CRIME_FACTOR / 20.0, 1.0)); }
    }
    if(FIRE_PROB_DENOMINATOR > 0 % game.FiresEnabled > 0) 
    {
        Random(random_is_fire);
        random_is_fire = random_is_fire % FIRE_PROB_DENOMINATOR;
        if(random_is_fire ? 1)
        {
            Random(random_building_id);
            random_building_id = random_building_id % n;
            building.GetDataByIndex(random_building_id);
            vector = building.vPosition;
            if(building.fPercFinished > 0.9999)
            {
                Building_StartFire(random_building_id);
                Notification_CreateNewStringPic("Civic Unrest", "Lack of civic services led to rioting. A building is on fire.", "services_icon.png", vector);
            }
        }
    }
    returnVoid();
}

defineFunction(BillAccounts, void, float:rub_to_bill, float:usd_to_bill)
{
    strcpy(tag_buf, rub_to_bill); atoi(tag_buf, int_rub_to_bill);
    strcpy(tag_buf, usd_to_bill); atoi(tag_buf, int_usd_to_bill);

    strcpy(tag_buf, money_rub); atof(tag_buf, float_money_rub);
    float_cost_rub = 0.0 - rub_to_bill;

    if (float_cost_rub < MaxFloat(float_money_rub, 0.0) & CAN_IMPORT_SERVICES > 0)
    {
        Money_AddRUB(int_rub_to_bill);
    }

    strcpy(tag_buf, money_usd); atof(tag_buf, float_money_usd);
    float_cost_usd = 0.0 - usd_to_bill;

    if (float_cost_usd < MaxFloat(float_money_usd, 0.0) & CAN_IMPORT_SERVICES > 0)
    {
        Money_AddUSD(int_usd_to_bill);
    }
    
    rub_sum2 = rub_sum2 + rub_to_bill;
    usd_sum2 = usd_sum2 + usd_to_bill;
    returnVoid();
}

defineFunction(SendToMarket, void, float:ServicesToMarket)
{
    current_price_rub = Price(1.0, ServicesToMarket);
    current_price_usd = Price(0.0, ServicesToMarket);
    
    money_rub_add = MaxFloat(ServicesToMarket, 0.0) * market_multiplier_rub * current_price_rub;
    money_usd_add = MaxFloat(ServicesToMarket, 0.0) * market_multiplier_usd * current_price_usd;
    
    money_rub_add = money_rub_add - MaxFloat(0.0 - ServicesToMarket, 0.0) * market_multiplier_rub * current_price_rub;
    money_usd_add = money_usd_add - MaxFloat(0.0 - ServicesToMarket, 0.0) * market_multiplier_usd * current_price_usd;

    BillAccounts(money_rub_add, money_usd_add);

    if (ServicesToMarket < 0.0)
    {

defineFunction(ProcessAdminWaterfall, void)
{
    if (BUREAUCRACY_ENABLED > 0)
    {
        CalculateSupplies();

        tick_supply_civic = supply_civic * time_factor;
        tick_supply_permit = supply_permit * time_factor;
        tick_supply_flex = supply_flex * time_factor;

        if (tick_supply_civic > tick_civic_demand) 
        {
            leftover_civic = tick_supply_civic - tick_civic_demand;
            civic_deficit = 0.0;
        } 
        else() 
        {
            leftover_civic = 0.0;
            civic_deficit = tick_civic_demand - tick_supply_civic;
            if (tick_supply_flex > civic_deficit) 
            {
                tick_supply_flex = tick_supply_flex - civic_deficit;
                civic_deficit = 0.0;
            } 
            else() 
            {
                civic_deficit = civic_deficit - tick_supply_flex;
                tick_supply_flex = 0.0;
            }
        }

        tick_permit_passive = permit_passive_demand * time_factor;
        tick_permit_active = permit_active_demand * time_factor;
        tick_permit_demand = tick_permit_passive + tick_permit_active;
        daily_permit_demand = daily_permit_demand + tick_permit_demand;

        if (tick_supply_permit > tick_permit_demand) 
        {
            leftover_permit = tick_supply_permit - tick_permit_demand;
            permit_deficit = 0.0;
        } 
        else() 
        {
            leftover_permit = 0.0;
            permit_deficit = tick_permit_demand - tick_supply_permit;
            if (tick_supply_flex > permit_deficit) 
            {
                tick_supply_flex = tick_supply_flex - permit_deficit;
                permit_deficit = 0.0;
            } 
            else() 
            {
                permit_deficit = permit_deficit - tick_supply_flex;
                tick_supply_flex = 0.0;
            }
        }

        civic_cache_capacity = demand_total * SUPPLY_CACHE_DAYS;
        permit_cache_capacity = permit_total_demand * PERMIT_CACHE_DAYS;

        if (civic_deficit > 0.0)
        {
            civic_cache = civic_cache - civic_deficit;
            if (civic_cache < 0.0)
            {
                unmitigated_civic = 0.0 - civic_cache;
                civic_cache = 0.0;
                daily_unmitigated_civic_deficit = daily_unmitigated_civic_deficit + unmitigated_civic;
            }
            else() { unmitigated_civic = 0.0; }
        }
        else() { unmitigated_civic = 0.0; }

        if (permit_deficit > 0.0)
        {
            permit_cache = permit_cache - permit_deficit;
            if (permit_cache < 0.0)
            {
                unmitigated_permit = 0.0 - permit_cache;
                permit_cache = 0.0;
                daily_unmitigated_permit_deficit = daily_unmitigated_permit_deficit + unmitigated_permit;
            }
            else() { unmitigated_permit = 0.0; }
        }
        else() { unmitigated_permit = 0.0; }

        export_surplus = 0.0;
        if (leftover_civic > 0.0)
        {
            civic_cache = civic_cache + leftover_civic;
            if (civic_cache > civic_cache_capacity)
            {
                export_surplus = export_surplus + (civic_cache - civic_cache_capacity);
                civic_cache = civic_cache_capacity;
            }
        }
        if (leftover_permit > 0.0)
        {
            permit_cache = permit_cache + leftover_permit;
            if (permit_cache > permit_cache_capacity)
            {
                export_surplus = export_surplus + (permit_cache - permit_cache_capacity);
                permit_cache = permit_cache_capacity;
            }
        }
        if (tick_supply_flex > 0.0)
        {
            civic_cache = civic_cache + tick_supply_flex;
            if (civic_cache > civic_cache_capacity)
            {
                tick_supply_flex = civic_cache - civic_cache_capacity;
                civic_cache = civic_cache_capacity;
                
                permit_cache = permit_cache + tick_supply_flex;
                if (permit_cache > permit_cache_capacity)
                {
                    export_surplus = export_surplus + (permit_cache - permit_cache_capacity);
                    permit_cache = permit_cache_capacity;
                }
            }
        }

        market_trade = export_surplus - unmitigated_civic;

        strcpy(tag_buf, market_trade); atoi(tag_buf, temp_trade_int); strcpy(tag_buf, temp_trade_int); atof(tag_buf, temp_trade_float);
        fraction_cache = fraction_cache + (market_trade - temp_trade_float);
        market_trade = temp_trade_float;

        if (fraction_cache < -1.0) { fraction_cache = fraction_cache + 1.0; market_trade = market_trade - 1.0; }
        elseif (fraction_cache > 1.0) { fraction_cache = fraction_cache - 1.0; market_trade = market_trade + 1.0; }

        SendToMarket(market_trade);
    }
    else()
    {
        supply = CalculateSupply();
        legacy_difference = supply - demand_total;
        legacy_difference = legacy_difference * time_factor;

        civic_cache_capacity = supply * SUPPLY_CACHE_DAYS;
        if (demand_total * SUPPLY_CACHE_DAYS > civic_cache_capacity) { civic_cache_capacity = demand_total * SUPPLY_CACHE_DAYS; }

        civic_cache = civic_cache + legacy_difference;

        if(civic_cache > civic_cache_capacity) { legacy_difference = civic_cache - civic_cache_capacity; civic_cache = civic_cache_capacity; }
        elseif(civic_cache < 0.0) { legacy_difference = civic_cache; civic_cache = 0.0; }
        else() { legacy_difference = 0.0; }

        if (legacy_difference > 0.0) { civic_cache = civic_cache + legacy_difference - ModuleInt(legacy_difference); legacy_difference = ModuleInt(legacy_difference); }
        elseif (legacy_difference < 0.0) { civic_cache = civic_cache + legacy_difference + ModuleInt(legacy_difference); legacy_difference = 0.0 - ModuleInt(legacy_difference); }

        if (civic_cache < 0.0) { civic_cache = civic_cache + 1.0; legacy_difference = legacy_difference - 1.0; }

        SendToMarket(legacy_difference);
    }
    returnVoid();
}

defineFunction(CorrectForTimeDilation, void)
{
    Script_EnterCriticalSection();
    Script_GetGameTime(game_time); 
    game_time = game_time - day_started_time; 

    if (game_time > 60 * 0.8) { game_speed = 1; time_factor = 0.0; }
    elseif(game_time > 20 * 0.8) { game_speed = 2; time_factor = (1.0 / (40.0/60.0)) - 1.0; }
    elseif(game_time > 5 * 0.8) { game_speed = 3; time_factor = (1.0 / (6.0/60.0)) - 1.0; }
    else() { game_speed = 4; time_factor = (1.0 / (2.0/60.0)) - 1.0; }

    if (game_speed > 2 & rub_sum2 ? 0.0 & usd_sum2 ? 0.0)
    {
        time_factor = -1.0;
        
        demand_total_vehicles = CalculateVehicleDemand();
        demand_total_vehicles = demand_total_vehicles * VEHICLE_DIFFICULTY; 
        demand_total_civics = CalculateCivicDemand();
        demand_total = demand_total_civics + demand_total_vehicles;
        tick_civic_demand = demand_total * time_factor;

        ProcessAdminWaterfall();
    }

    if (time_factor > 0.0)
    {
        rub_true_up = rub_sum2 * time_factor;
        usd_true_up = usd_sum2 * time_factor;
        BillAccounts(rub_true_up, usd_true_up);
    }

    Script_LeaveCriticalSection();
    returnVoid();
}
