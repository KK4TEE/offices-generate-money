defineFunction(CalculateVehicleDemand, float)
{
    Vehicle_GetNumberOfVehicles(m);
    vehicles_in_depot = 0;
    vehicles_num_trucks = 0;
    vehicles_num_trains = 0;
    vehicles_num_ships = 0;
    vehicles_num_airplanes = 0;
    demand_vehicles = 0.0;

    for (i=0, i<m, i=i+1)
    {
        vehicle.GetDataByIndex(i);
        vehicle_cost = 0.0;
        if(vehicle.nTransportType ? RESOURCE_TRANSPORT_PASSANGER)
        {
            //for passenger vehicles, the cost is determined per 10 passengers
            vehicle_cost = vehicle.fResourceCapacity / 10.0;
        }
        elseif(vehicle.nTransportType > -1 & vehicle.nTransportType < RESOURCE_TRANSPORT_NUM)
        {
            //for freight vehicles, the cost is determined per 1 t
            vehicle_cost = vehicle.fResourceCapacity;
        }

        // Set road vehicle cost based on VEHICLESKILL
        if (vehicle.nVehicleType > 0 & vehicle.nVehicleType < 3)
        {
            vehicles_num_trucks = vehicles_num_trucks + 1;

            if(vehicle.nSkill[0] ? VEHICLESKILL_SEEDING)
            {
                vehicle_cost = BASE_COST_SEEDING;
            }
            elseif(vehicle.nSkill[0] ? VEHICLESKILL_HARVESTING)
            {
                vehicle_cost = BASE_COST_HARVESTING;
            }
            elseif(vehicle.nSkill[0] ? VEHICLESKILL_SNOWPLOW)
            {
                vehicle_cost = BASE_COST_SNOWPLOWING;
            }
            elseif(vehicle.nSkill[0] ? VEHICLESKILL_PERSONAL)
            {
                vehicle_cost = BASE_COST_PERSONAL;
            }
            else()
            {
            // Finally, if no skill set by capacity
            vehicle_cost = BASE_COST_ROAD + vehicle_cost * COST_ROAD_FACTOR;
            }
        }

        //trains
        elseif(vehicle.nVehicleType < 6)
        {
            vehicles_num_trains = vehicles_num_trains + 1;
            vehicle_cost = BASE_COST_RAIL + COST_RAIL_FACTOR * vehicle_cost;
        }
        //ships
        elseif(vehicle.nVehicleType ? VEHICLETYPE_SHIP)
        {
            vehicles_num_ships = vehicles_num_ships + 1;
            vehicle_cost = vehicle_cost + BASE_COST_SHIP + COST_SHIP_FACTOR * vehicle_cost;
        }
        //planes, helicopters
        elseif(vehicle.nVehicleType ? VEHICLETYPE_AIRPLANE | vehicle.nVehicleType ? VEHICLETYPE_HELICOPTER)
        {
            vehicles_num_planes = vehicles_num_planes + 1;
            vehicle_cost = vehicle_cost + BASE_COST_AIR + COST_AIR_FACTOR * vehicle_cost;
        }

        // If the vehicle is idle in a depot, it doesn't need many services
        // Vehicles on a route report homeWorkplaceID = 0
        // TODO: Add check if it is an endstation

        if(vehicle.nBuilding_CurrentID ? vehicle.nBuilding_HomeWorkplaceID & vehicle.nBuilding_CurrentID > 0)
        {
            debug_workplace_id = vehicle.nBuilding_HomeWorkplaceID;
            vehicle_cost = vehicle_cost * VEHICLE_IDLE_COST;
            vehicles_in_depot = vehicles_in_depot + 1;
        }

        demand_vehicles = demand_vehicles + vehicle_cost;
    }
    return(demand_vehicles);
}

defineFunction(CalculateCivicDemand, float)
{
     population = 0;
    for (i=0, i<n, i=i+1)
    {
        building.GetDataByIndex(i);
        if(building.nType ? BUILDINGTYPE_LIVING & building.fPercFinished > 0.9999)
        {
            Building_IsOldTownActive(i, is_active);
            population = population + is_active * is_active * building.nWorkersNum;
        }
    }
    demand = 0.0;
    if(!(population > GRACE_CITIZENS))
    {
        demand = population * INTERNAL_DEMAND_FACTOR * GRACE_FACTOR;
    }
    else()
    {
        demand = INTERNAL_DEMAND_FACTOR * (GRACE_CITIZENS * GRACE_FACTOR + population - GRACE_CITIZENS);
    }

    return(demand);
}

defineFunction(CalculateSupply, float)
{
    supply = 0.0;
    for (i=0, i<n, i=i+1)
    {
        services = 0.0;
        building.GetDataByIndex(i);

        if (!(building.nSubType ? BUILDINGSUBTYPE_ROAD) & building.fPercFinished > 0.9999)
        {
            if(building.nType ? BUILDINGTYPE_CITYHALL)
            {
                // For each worker currently in the building, add some services
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    // Obtain the worker
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);

                    //  The money we get from uneducated workers is diminished
                    //  while the number of educated is increased
                    service_scale = PRIMARY_ED_FACTOR;
                    if (worker.fEducation>1.999) {
                        service_scale = UNIVERSITY_ED_FACTOR;
                    }
                    services = services + service_scale;
                }
            }

            if(building.nType ? BUILDINGTYPE_UNIVERSITY & building.nSubType ? BUILDINGSUBTYPE_SOVIET)
            {
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    if (worker.bWorkplaceAsProfesor ? 1) {
                        services = services + HQ_FACTOR;
                    }
                }
            }

            if(building.nType ? BUILDINGTYPE_SECRET_POLICE & building.nVehiclesWorkingNum < 1)
            {
                // Only count this building towards supply if it has no vehicles to perform duties with
                // For each worker currently in the building, add some services
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    // Obtain the worker
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);

                    //  The money we get from uneducated workers is diminished
                    //  while the number of educated is increased
                    service_scale = PRIMARY_ED_FACTOR;
                    if (worker.fEducation>1.999) {
                        service_scale = UNIVERSITY_ED_FACTOR;
                    }
                    services = services + service_scale;
                }
            }

            if (VEHICLE_DIFFICULTY > 0.0 & building.nType ? BUILDINGTYPE_AIRPLANE_TOWER)
            {
                // For each worker currently in the building, add some services
                for (j=0, j<building.nWorkersNum, j=j+1)
                {
                    // Obtain the worker
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);

                    //  The money we get from uneducated workers is diminished
                    //  while the number of educated is increased
                    service_scale = PRIMARY_ED_FACTOR;
                    if (worker.fEducation>1.999) {
                        service_scale = UNIVERSITY_ED_FACTOR;
                    }
                    services = services + service_scale;
                }
            }
        }

        if(building.fEletric_Voltage < 0.9 & game.EletricityLevel > 0)
        {
            services = services * NO_POWER_FACTOR;
        }
        if(building.fWater_CurrentDrinkAmount < 0.001 & game.WaterManagement > 0)
        {
            services = services * NO_WATER_FACTOR;
        }
        supply = supply + services;
    }
    return(supply);
}

defineFunction(CalculatePermits, void)
{
    permit_supply = 0.0;
    permit_demand = 0.0;
    police_bonus = 0.0;
    
    // Reset our new tracking variables
    count_active_sites = 0;
    count_construction_offices = 0;
    supply_city_hall = 0.0;
    supply_university = 0.0;
    demand_overhead = 0.0;
    demand_active = 0.0;
    
    Building_GetNumberOfBuildings(n);

    for (i = 0, i < n, i = i + 1)
    {
        building.GetDataByIndex(i);

        // --- SUPPLY ---
        if (building.fPercFinished > 0.9999)
        {
            if (building.nType ? BUILDINGTYPE_CITYHALL)
            {
                supply_city_hall = supply_city_hall + (building.nWorkersNum * PERMIT_SUPPLY_CITYHALL);
            }
            elseif (building.nType ? BUILDINGTYPE_UNIVERSITY & building.nSubType ? BUILDINGSUBTYPE_TECHNICAL)
            {
                for (j = 0, j < building.nWorkersNum, j = j + 1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);
                    
                    if (worker.bWorkplaceAsProfesor ? 1)
                    {
                        supply_university = supply_university + PERMIT_SUPPLY_TECH_PROF;
                    }
                    else()
                    {
                        supply_university = supply_university + PERMIT_SUPPLY_TECH_STAFF;
                    }
                }
            }
            elseif (building.nType ? BUILDINGTYPE_SECRET_POLICE)
            {
                police_bonus = police_bonus + (building.nWorkersNum * PERMIT_SUPPLY_SECRET_POLICE);
            }
            
            // Overhead Demand
            if (building.nType ? BUILDINGTYPE_CONSTRUCTION_OFFICE | building.nType ? BUILDINGTYPE_CONSTRUCTION_OFFICE_RAIL)
            {
                demand_overhead = demand_overhead + PERMIT_DEMAND_OFFICE; 
                count_construction_offices = count_construction_offices + 1;
            }
        }

        // --- ACTIVE DEMAND ---
        if (building.bConstructionActive ? 1)
        {
            demand_active = demand_active + (building.nWorkersNeeded * PERMIT_DEMAND_ACTIVE_MULT); 
            count_active_sites = count_active_sites + 1;
        }
    }

    // Tally up the final numbers
    permit_supply = supply_city_hall + supply_university;
    permit_supply = permit_supply * (1.0 + police_bonus);
    
    permit_demand = demand_overhead + demand_active;

    if (permit_demand > permit_supply)
    {
        permit_deficit = permit_demand - permit_supply;
    }
    else()
    {
        permit_deficit = 0.0;
    }

    returnVoid();
}

defineFunction(ProcessPermitPenalties, void)
{
    if (permit_demand > 0.0)
    {
        permit_deficit_ratio = (permit_deficit / permit_demand) * 100.0; 
    }
    else()
    {
        permit_deficit_ratio = 0.0;
    }

    Building_GetNumberOfBuildings(n);
    
    for (i = 0, i < n, i = i + 1)
    {
        building.GetDataByIndex(i);
        Building_GetCharTag(i, tag_buf);

        // Initial tagging logic
        strcmp(tag_buf, tag_empty, cmp_res);
        if (cmp_res ? 0)
        {
            if (building.fPercFinished > 0.9999)
            {
                Building_SetTagChar(tag_clean, i); 
            }
            else()
            {
                Building_SetTagChar(tag_under_construction, i); 
            }
        }

        // PENALTY A: Fires
        if (building.bConstructionActive ? 1 & permit_deficit > 0.0)
        {
            Random(roll);
            roll = roll % 10000;
            float_roll = roll * 1.0; 
            
            if (float_roll < (permit_deficit_ratio * PERMIT_FIRE_CHANCE_MULT) & building.bFireEnabled ? 1)
            {
                Building_StartFire(i);
            }
        }

        // PENALTY B: Rushed Builds / The Curse (Completion Check)
        if (building.fPercFinished > 0.9999)
        {
            strcmp(tag_buf, tag_under_construction, cmp_res);
            if (cmp_res ? 0)
            {
                if (permit_deficit > 0.0)
                {
                    Building_SetTagChar(tag_rushed, i); 
                }
                else()
                {
                    Building_SetTagChar(tag_clean, i); 
                }
            }

            // Passive Status Drain
            strcmp(tag_buf, tag_rushed, cmp_res);
            if (cmp_res ? 0)
            {
                for (j = 0, j < building.nWorkersNum, j = j + 1)
                {
                    Building_Workers_GetID(i, j, worker_id);
                    worker.GetDataByIndex(worker_id);

                    if (worker.fStatusHealth > 0.1)
                    {
                        Person_SetStatus(worker_id, 2, worker.fStatusHealth - PERMIT_PENALTY_HEALTH_DRAIN);
                    }
                    if (worker.fStatusHappiness > 0.1)
                    {
                        Person_SetStatus(worker_id, 0, worker.fStatusHappiness - PERMIT_PENALTY_HAPP_DRAIN);
                    }
                }
            }
        }
    }
    returnVoid();
}

//currency: 1 - rubles, 0 - dollars, excess: >0 - export, <0 - import
defineFunction(Price, float, int:currency, float:excess) 
{
    price_inertia = supply + BASE_FOREIGN_SUPPLY_DEMAND + FOREIGN_SUPPLY_DEMAND_FACTOR * population;
    factor = 1.0 - excess / price_inertia;
    if(excess > 0.0)
    {
        price = factor * (currency * PRICE_SELL_RUB + (1 - currency) * PRICE_SELL_USD) / (100 ^ NUMBER_OF_PRICE_DECREASES);
    }
    else()
    {
        price = factor * (currency * PRICE_BUY_RUB + (1 - currency) * PRICE_BUY_USD) / (100 ^ NUMBER_OF_PRICE_DECREASES);
    }
    return(price);
}

defineFunction(MakePeopleAngry, void, float:unpaid_services)
{
    for(i=0, i<population, i=i+1)
    {
        worker.GetDataByIndex(i);
        Person_SetStatus(i, 0, MaxFloat(worker.fStatusHappiness - unpaid_services * UNHAPPINESS_FACTOR / 20.0, 0.0)) //decrease happiness
        Person_SetStatus(i, 3, MaxFloat(worker.fStatusSoviet - unpaid_services * LOYALTY_FACTOR / 20.0, 0.0)) //decrease loyalty
        if(game.CrimeEnabled > 0)
        {
            Person_SetStatus(i, 10, MinFloat(worker.fStatusCrime + unpaid_services * CRIME_FACTOR / 20.0, 1.0)) //increase crime
        }
    }
    //increased probability of fire
    if(FIRE_PROB_DENOMINATOR > 0 % game.FiresEnabled > 0) 
    {
        Random(random_is_fire);
        random_is_fire = random_is_fire % FIRE_PROB_DENOMINATOR;
        if(random_is_fire ? 1)
        {
            Random(random_building_id);
            random_building_id = random_building_id % n;
            building.GetDataByIndex(random_building_id);
            vector = building.vPosition;
            if(building.fPercFinished > 0.9999)
            {
                Building_StartFire(random_building_id);
                Notification_CreateNewStringPic(
                "Insufficient services",
                "Lack of overseeing has led to an incident in one of our buildings. The building is on fire.",
                "services_icon.png",
                vector);
            }
        }
    }
    returnVoid();
}

defineFunction(BillAccounts, void, float:rub_to_bill, float:usd_to_bill)
{
    // Engine functions Money_AddRUB/USD strictly require ints (Type 2).
    // This assignment will trigger a compiler warning, but it prevents the crash.
    int_rub_to_bill = rub_to_bill; 
    int_usd_to_bill = usd_to_bill;

    if (-1.0 * rub_to_bill < MaxInt(money_rub, 0) & CAN_IMPORT_SERVICES > 0)
    {
        Money_AddRUB(int_rub_to_bill);
    }
    if (-1.0 * usd_to_bill < MaxInt(money_usd, 0) & CAN_IMPORT_SERVICES > 0)
    {
        Money_AddUSD(int_usd_to_bill);
    }
    
    // We keep tracking the daily sum accurately using the precise floats
    rub_sum2 = rub_sum2 + rub_to_bill;
    usd_sum2 = usd_sum2 + usd_to_bill;

    returnVoid();
}

defineFunction(SendToMarket, void, float:ServicesToMarket)
{
    current_price_rub = ScaledRUB(Price(1.0, ServicesToMarket));
    current_price_usd = ScaledUSD(Price(0.0, ServicesToMarket));
    money_rub_add = MaxFloat(ServicesToMarket, 0.0) * market_multiplier_rub * current_price_rub;
    money_usd_add = MaxFloat(ServicesToMarket, 0.0) * market_multiplier_usd * current_price_usd;
    money_rub_add = money_rub_add - MaxFloat(-ServicesToMarket, 0.0) * market_multiplier_rub * current_price_rub;
    money_usd_add = money_usd_add - MaxFloat(-ServicesToMarket, 0.0) * market_multiplier_usd * current_price_usd;

    BillAccounts(money_rub_add, money_usd_add);

    // Restored penalty for unfulfilled demand if importing fails
    if (ServicesToMarket < 0.0)
    {
        if (-1.0 * money_rub_add < MaxInt(money_rub, 0) & CAN_IMPORT_SERVICES > 0)
        {
            // Successful RUB import, no penalty
        }
        else()
        {
            if (market_multiplier_rub > 0.0)
            {
                MakePeopleAngry(market_multiplier_rub * ModuleFloat(ServicesToMarket * 1.0 / demand_total));
            }
        }

        if (-1.0 * money_usd_add < MaxInt(money_usd, 0) & CAN_IMPORT_SERVICES > 0)
        {
            // Successful USD import, no penalty
        }
        else()
        {
            if (market_multiplier_usd > 0.0)
            {
                MakePeopleAngry(market_multiplier_usd * ModuleFloat(ServicesToMarket * 1.0 / demand_total));
            }
        }
    }

    returnVoid();
}

defineFunction(CorrectForTimeDilation, void)
{
    // In-game time is scaled and reported differently than wall time
    Script_EnterCriticalSection();
    Script_GetGameTime(game_time); // Refresh game time upon entering the critical section
    game_time = game_time - day_started_time; // The scaled time reported by the game engine

    if (game_time > 60 * 0.8)
    {
        // Normal game speed
        // No balance true-up required
        game_speed = 1;
        time_factor = 0.0;
    }
    elseif(game_time > 20 * 0.8)
    {
        // Fast game speed
        game_speed = 2;
        time_factor = (1.0 / (40.0/60.0)) - 1.0;
    }
    elseif(game_time > 5 * 0.8)
    {
         // Num1 debug game speed
        game_speed = 3;
        time_factor = (1.0 / (6.0/60.0)) - 1.0;
    }
    else()
    {
        // Num2 debug game speed
        game_speed = 4;
        time_factor = (1.0 / (2.0/60.0)) - 1.0;
    }

    // Fallback calculation
        if (game_speed > 2 & rub_sum2 ? 0.0 & usd_sum2 ? 0.0)
        {
            time_factor = -1.0;
            demand_total_vehicles = 0.0;
            if (VEHICLE_DIFFICULTY > 0.0)
            {
                demand_total_vehicles = 0.0 + CalculateVehicleDemand();
            }

            demand_total_civics = CalculateCivicDemand();
            demand_total = demand_total_civics + demand_total_vehicles;
            supply = CalculateSupply();
            difference = supply - demand_total;
            SendToMarket(difference);
        }

    if (time_factor > 0.0)
    {
        rub_true_up = rub_sum2 * time_factor;
        usd_true_up = usd_sum2 * time_factor;
        BillAccounts(rub_true_up, usd_true_up);
    }

    Script_LeaveCriticalSection();
    returnVoid();
}

defineFunction(Money, void, float:rub, float:usd)
{
    InitConstants();
    InitSettings();
    NEWLINE = 10; // ASCII LF
    //check correctness of the constants in the settings file
    INFLATION_RUB = MaxFloat(INFLATION_RUB, 1.0);
    INFLATION_USD = MaxFloat(INFLATION_USD, 1.0);
    PRICE_BUY_RUB = MaxFloat(PRICE_BUY_RUB, 0.0);
    PRICE_BUY_USD = MaxFloat(PRICE_BUY_USD, 0.0);
    PRICE_SELL_RUB = MaxFloat(PRICE_SELL_RUB, 0.0);
    PRICE_SELL_USD = MaxFloat(PRICE_SELL_USD, 0.0);
    FOREIGN_SUPPLY_DEMAND_FACTOR = MaxFloat(FOREIGN_SUPPLY_DEMAND_FACTOR, 0.0);
    PRIMARY_ED_FACTOR = MaxFloat(PRIMARY_ED_FACTOR, 0.0);
    UNIVERSITY_ED_FACTOR = MaxFloat(UNIVERSITY_ED_FACTOR, 0.0);
    year = START_YEAR;
    rub_sum = 0.0;
    usd_sum = 0.0;
    market_multiplier_rub = rub;
    market_multiplier_usd = usd;
    Script_GetGameTime(game_time);
    Script_GetGameTime(game_time_prev);
    
    //deduce the start year
    if(START_YEAR ? 0)
    {
        Date_GetCurrentDate_D365Y(day, year);
        i=1799;
        while(i<year+1 & START_YEAR ? 0)
        {
            stat_record.GetFromDateToDate_D365Y(0, i, 0, i+1);
            value = stat_record.Vehicles_ImportUSD + stat_record.Vehicles_ImportRUB + stat_record.ImportTonsRUB.gravel + stat_record.ImportTonsUSD.gravel;
            if(value > 0.0)
            {
                START_YEAR = i;
            }
            i=i+1;
        }
        if(START_YEAR ? 0)
        {   
            START_YEAR = year;
        }
    }

    // --- BUREAUCRACY MOD: INITIALIZE TAGS ---
    strcpy(tag_under_construction, "uc");
    strcpy(tag_rushed, "rushed_build");
    strcpy(tag_clean, "clean_build");
    strcpy(tag_empty, "");

    while (1 ? 1)
    {
        game.GetCurrentGameSettigns();
        rub_sum2 = 0.0;
        usd_sum2 = 0.0;
        clicked_object_id = -1;
        Date_GetCurrentDate_D365Y(current_day, year);
        Date_GetCurrentDate_D365Y(processing_day, year);
        Script_GetGameTime(day_started_time);
        while(current_day ? processing_day)
        {
            Date_GetCurrentDate_D365Y(current_day, year);
            Script_GetGameTime(game_time);
            if(game_time - game_time_prev > 2.5)
            {
                Script_EnterCriticalSection();
                time_factor = (game_time - game_time_prev) / 60.0;
                Building_GetNumberOfBuildings(n);
                Money_GetRUB(money_rub);
                Money_GetUSD(money_usd);

                if (game_speed ? 1 | game_speed ? 2)
                {
                    demand_total_vehicles = 0.0;
                    if (VEHICLE_DIFFICULTY > 0.0)
                    {
                        demand_total_vehicles = 0.0 + CalculateVehicleDemand();
                    }

                    demand_total_civics = CalculateCivicDemand();
                    demand_total = demand_total_civics + demand_total_vehicles;

                    supply = CalculateSupply();
                    difference = supply - demand_total;

                    difference = difference * time_factor;

                    // Grow the cache if supply or demand is a historic high
                    if (supply * SUPPLY_CACHE_DAYS > supply_cache_capacity)
                    {
                        supply_cache_capacity = supply * SUPPLY_CACHE_DAYS;
                    }
                    if (demand_total * SUPPLY_CACHE_DAYS > supply_cache_capacity)
                    {
                        supply_cache_capacity = demand_total * SUPPLY_CACHE_DAYS;
                    }

                    // Determine how the service cache should be used
                    supply_cache = supply_cache + difference;

                    if(supply_cache > supply_cache_capacity)
                    {
                        difference = supply_cache - supply_cache_capacity;
                        supply_cache = supply_cache_capacity;
                    }
                    elseif(supply_cache < 0.0)
                    {
                        difference = supply_cache;
                        supply_cache = 0.0;
                    }
                    else()
                    {
                        // All transactions completed using the cache. No trade needed.
                        difference = 0.0;
                    }

                    if (difference > 0.0)
                    {
                        supply_cache = supply_cache + difference - ModuleInt(difference);  
                        difference = ModuleInt(difference); 
                    }
                    elseif (difference < 0.0)
                    {
                        supply_cache = supply_cache + difference + ModuleInt(difference);  
                        difference = 0.0 - ModuleInt(difference);  
                    }

                    if (supply_cache < 0.0)
                    {
                        supply_cache = supply_cache + 1.0;
                        difference = difference - 1.0;
                    }

                    SendToMarket(difference);
                }

                // --- BUREAUCRACY MOD: REAL-TIME PERMIT CALCULATION FOR UI ---
                if (BUREAUCRACY_ENABLED > 0)
                {
                    CalculatePermits(); 
                }

                // User Interface
                difference = supply - demand_total; // Rescale for full day
                HandleStatisticsWindow();
                UpdateInformationDisplay();

                game_time_prev = game_time;
                Script_LeaveCriticalSection();
            }
            Script_Sleep(0.1);
        }

        // The while loop has broken, so the day has changed.
        CorrectForTimeDilation();

        // --- BUREAUCRACY MOD: DAY ROLLOVER PENALTIES ---
        if (BUREAUCRACY_ENABLED > 0)
        {
            Script_EnterCriticalSection();
            ProcessPermitPenalties(); 
            Script_LeaveCriticalSection();
        }

        rub_sum = rub_sum2;
        usd_sum = usd_sum2;
    }
    returnVoid();
}
