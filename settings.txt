defineFunction(InitSettings, void)
{
    // You can modify the following values to enable or disable features or change the balance.
    // Please keep at least one digit after the decimal point if present.
    
    // =========================================================================
    // CORE SCRIPT & UI SETTINGS
    // =========================================================================
    INFORMATION_SHOWN = 2; // 2 - full information (3 lines), 1 - concise information (1 line), 0 - turned off
    START_YEAR = 0; // Starting year of the game. If 0, the script will try to determine it based on import history.
    DEBUG_MODE_ENABLED = 1; // Set to 1 to print debug text in the generated UI reports. 0 disables the extra text.

    // =========================================================================
    // SERVICES & INTERNAL MARKET SETTINGS
    // =========================================================================
    INFLATION_RUB = 1.09; // Rubles inflation rate
    INFLATION_USD = 1.027; // Dollars inflation rate
    INTERNAL_DEMAND_FACTOR = 0.02; // Workers needed in offices to serve 1 citizen (0.0 or less disables feature)
    GRACE_CITIZENS = 0; // Demand for services is decreased for your first GRACE_CITIZENS non-dormant citizens
    GRACE_FACTOR = 0.5; // Decrease in demand for your first GRACE_CITIZENS non-dormant citizens
    SUPPLY_CACHE_DAYS = 3; // Days of supply to cache to smooth out commuting disruptions.
    NUMBER_OF_PRICE_DECREASES = 0; // How many times you've used cheats to divide prices by 100
    CAN_IMPORT_SERVICES = 1; // 1 or more allows importing services; 0 or less forces domestic generation

    // Base prices of 1 unit of services
    PRICE_BUY_RUB = 8.0;
    PRICE_BUY_USD = 10.0;
    PRICE_SELL_RUB = 2.0;
    PRICE_SELL_USD = 2.5;
    
    BASE_FOREIGN_SUPPLY_DEMAND = 200.0; // Foreign supply/demand inertia for calculating market price
    FOREIGN_SUPPLY_DEMAND_FACTOR = 0.03; // How much your population increases the foreign supply and demand
    
    PRIMARY_ED_FACTOR = 1.0; // Services generated by a worker with primary education
    UNIVERSITY_ED_FACTOR = 1.75; // Services generated by a worker with university education
    HQ_FACTOR = 0.75; // Services generated by a university educated worker in a party HQ
    
    // The following constants determine the negative influence of unfulfilled service needs
    UNHAPPINESS_FACTOR = 0.005; // -0.5 percent points per day
    LOYALTY_FACTOR = 0.0025; // -0.25 percent points per day
    CRIME_FACTOR = 0.005; // +0.5 percent points per day
    FIRE_PROB_DENOMINATOR = 100; // Probability per tick of fire is 1 over this (0 disables)
    
    NO_POWER_FACTOR = 0.25; // Generation scaled when office lacks electricity 
    NO_WATER_FACTOR = 0.5; // Generation scaled when office lacks water 

    // =========================================================================
    // VEHICLE ADMINISTRATIVE OVERHEAD
    // =========================================================================
    VEHICLE_DIFFICULTY = 1.0; // Scale factor for vehicle costs. 0.0 disables. 0.5 halves supply use.
    VEHICLE_IDLE_COST = 0.01; // Multiplier for vehicles idle in a depot

    BASE_COST_ROAD = 1.0;
    BASE_COST_RAIL = 2.0;
    BASE_COST_SHIP = 10.0;
    BASE_COST_AIR = 4.0;
    
    BASE_COST_SEEDING = 1.25;
    BASE_COST_HARVESTING = 1.25;
    BASE_COST_SNOWPLOWING = 1.5;
    BASE_COST_PERSONAL = 0.02;
    
    // Administrative cost needed by each vehicle per 1t of resources or 10 passengers
    COST_ROAD_FACTOR = 0.02;
    COST_RAIL_FACTOR = 0.002;
    COST_SHIP_FACTOR = 0.0002;
    COST_AIR_FACTOR = 0.05;

    // =========================================================================
    // BUREAUCRACY & CONSTRUCTION PERMITS MOD
    // =========================================================================
    
    // Master Toggle: 1 to enable the Bureaucracy mod, 0 to completely disable it.
    BUREAUCRACY_ENABLED = 1; 

    // --- Supply Generation (Admin Pool) ---
    // Permits generated per generic worker in a City Hall
    PERMIT_SUPPLY_CITYHALL = 1.0; 
    
    // Permits generated per staff member (non-professor) at a Technical University
    PERMIT_SUPPLY_TECH_STAFF = 1.5; 
    
    // Permits generated per Professor at a Technical University
    PERMIT_SUPPLY_TECH_PROF = 3.0; 
    
    // Percentage efficiency boost to total admin pool per Secret Police worker (0.005 = 0.5% boost)
    PERMIT_SUPPLY_SECRET_POLICE = 1.5; 
	PERMIT_SUPPLY_REGULAR_POLICE = 1.0;

    // --- Demand Generation ---
    // Passive daily permit drain just for having a Construction Office or Rail Office built
    PERMIT_DEMAND_OFFICE = 5.0; 
    
    // Multiplier for active construction demand based on total max workers needed for the project
    // Example: 0.1 means a building requiring 100 workers demands 10 permits daily.
    PERMIT_DEMAND_ACTIVE_MULT = 0.1; 

    // --- Buffer & Penalties ---
    // How many days of active construction demand the Permit Cache can hold to smooth out fluctuations
    PERMIT_CACHE_DAYS = 3.0;

    // Daily drain on health for citizens living/working in a permanently "rushed" building
    PERMIT_PENALTY_HEALTH_DRAIN = 0.005; 
    
    // Daily drain on happiness for citizens living/working in a permanently "rushed" building
    PERMIT_PENALTY_HAPP_DRAIN = 0.005; 
    
    // Multiplier for the chance of a fire breaking out on an active site during an unmitigated deficit
    // 1.0 uses the default ratio (e.g., 50% deficit = up to 0.5% chance per tick)
    PERMIT_FIRE_CHANCE_MULT = 1.0; 

    returnVoid();
}
